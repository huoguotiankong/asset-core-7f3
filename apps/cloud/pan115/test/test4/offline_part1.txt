function base32ToHex(s) {
    let alphabet = "ABCDEFGHIJKLMNOPQRSTUVWXYZ234567";
    let bits = "";
    let out = "";
    let t = String(s || "").toUpperCase().replace(/=+$/g, "");
    for (let i = 0; i < t.length; i++) {
        let n = alphabet.indexOf(t.charAt(i));
        if (n < 0) return "";
        let b = n.toString(2);
        bits += ("00000" + b).slice(-5);
    }
    for (let j = 0; j + 8 <= bits.length; j += 8) {
        out += ("0" + parseInt(bits.slice(j, j + 8), 2).toString(16)).slice(-2);
    }
    return out.length === 40 ? out.toLowerCase() : "";
}

function btihOf(s) {
    let raw = String(s || "");
    try { raw = decodeURIComponent(raw); } catch (e) { }
    let m = /(?:^|[?&])xt=urn:btih:([A-Za-z0-9]+)/i.exec(raw);
    if (!m) return "";
    let h = String(m[1] || "");
    if (/^[0-9a-f]{40}$/i.test(h)) return h.toLowerCase();
    if (/^[A-Z2-7]{32}$/i.test(h)) return base32ToHex(h);
    return "";
}

function listTasks(maxPages) {
    let out = [], seen = {};
    for (let p = 1; p <= maxPages; p++) {
        let r = client.listOfflineTask(p);
        let a = (r && r.tasks) || [];
        if (!a.length) break;
        let sig = a.length + "|" + (a[0] ? a[0].infoHash : "");
        if (seen[sig]) break;
        seen[sig] = true;
        a.forEach((x) => out.push(x));
        if (r.pageCount && p >= r.pageCount) break;
    }
    return out;
}

function findSameTask(tasks, link) {
    let raw = String(link || "").trim();
    let hh = btihOf(raw);
    for (let i = 0; i < tasks.length; i++) {
        let t = tasks[i] || {};
        if (hh && String(t.infoHash || "").toLowerCase() === hh) return t;
        if (raw && String(t.url || "").trim() === raw) return t;
    }
    return null;
}
