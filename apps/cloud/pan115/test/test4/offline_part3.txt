function cachedResultUrl(cached, hash, ruleTitle) {
    if (!cached) return "";
    if (cached.mode === "single" && cached.fid) {
        return $().lazyRule((cacheText, cacheKey) => {
            let api2 = $.require("115Api");
            let c = api2.newClient();
            let x;
            try { x = JSON.parse(cacheText); } catch (e) { return "toast://播放缓存损坏"; }
            try {
                let f = c.getFile(String(x.fid || ""));
                if (!f || f.isDirectory) {
                    setItem(cacheKey, "");
                    return "toast://缓存文件已失效，请重新进入磁链";
                }
                if (!f.pickCode && x.pc) f.pickCode = x.pc;
                return api2.player.resolve(JSON.stringify({
                    pc: f.pickCode || "",
                    fid: f.fileId || "",
                    name: f.name || x.name || "",
                    kind: "video"
                }));
            } catch (e2) {
                setItem(cacheKey, "");
                return "toast://缓存文件已失效，请重新进入磁链";
            }
        }, JSON.stringify(cached), resultCacheKey(hash));
    }
    if (cached.mode === "multi" && cached.videos && cached.videos.length) {
        return "hiker://page/115OfflineResult?rule=" + encodeURIComponent(ruleTitle) +
            "&page=fypage&fid=" + encodeURIComponent(String(cached.rootFid || "")) +
            "&pid=" + encodeURIComponent(String(cached.rootPid || "")) +
            "&hash=" + encodeURIComponent(String(hash || "")) +
            "&name=" + encodeURIComponent(String(cached.taskName || "离线结果"));
    }
    return "";
}
